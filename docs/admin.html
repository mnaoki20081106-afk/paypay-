▸ セットアップ手順 
# 1. Tor インストール
apt install -y tor

# 2. /etc/tor/torrc に追記
ControlPort 9051
CookieAuthentication 1

# 3. 起動 & 自動起動
service tor start
systemctl enable tor

# 4. 環境変数（変更する場合のみ）
TOR_SOCKS_PROXY=socks5h://127.0.0.1:9050
TOR_CONTROL_HOST=127.0.0.1
TOR_CONTROL_PORT=9051
TOR_CONTROL_PASSWORD=   # CookieAuthentication なら空欄で OK
Cloudflare Worker 設定
▸ デプロイ手順 
# 1. wrangler インストール
npm install -g wrangler
wrangler login

# 2. KV ネームスペース作成
wrangler kv namespace create CREDS
# → id をメモ

# 3. wrangler.toml を作成（worker.js の隣）
name = "paypay-store"
main = "worker.js"
compatibility_date = "2024-01-01"

[[kv_namespaces]]
binding = "CREDS"
id = "<上でメモした id>"

# 4. シークレット登録
wrangler secret put SECRET
# → 任意の文字列を入力（X-Secret に設定するもの）

# 5. デプロイ
wrangler deploy
# → https://paypay-store.xxx.workers.dev が発行される
